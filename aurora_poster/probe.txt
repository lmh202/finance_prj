Paste this into the devtools console with poster.html open, or drop it in
temporarily just before </body> and reload. Every panel is overflow:hidden, so
content that does not fit is clipped with no error, no console warning and no
visible glitch — it simply vanishes, and it will not show up in a quick visual
scan either. Require over=0 on all seven panels AND xover=0 on every strip
before exporting.

The second list matters as much as the first: the ledger trace, the tech-strip
pills and the contract rows are all white-space:nowrap, so they overflow
sideways rather than wrapping. A clipped chip at the right edge of the ledger
looks exactly like a chip that was never there.

<div id="probe" style="position:fixed;z-index:99;left:0;bottom:0;background:#000;color:#0f0;font:11px monospace;padding:6px;max-width:100%"></div>
<script>
window.addEventListener('load', function () {
  var mm = 1 / 3.779528, r = [];

  document.querySelectorAll('.panel').forEach(function (p, i) {
    var h = p.querySelector('.h-section');
    var n = h ? h.textContent.replace(/\s+/g, ' ').trim().slice(0, 24) : 'panel' + i;
    var o = p.scrollHeight - p.clientHeight;
    r.push('PANEL~' + n +
           '~box=' + (p.clientHeight * mm).toFixed(0) +
           '~needs=' + (p.scrollHeight * mm).toFixed(0) +
           '~over=' + (o > 1 ? (o * mm).toFixed(1) : '0') + '~');
  });

  // Horizontal clipping. These are the containers whose children are nowrap.
  '.tech-strip,.ledger,.ledger-row,.sigs,.steps,.outs,.contract,.bars,.close-grid'
    .split(',').forEach(function (sel) {
      document.querySelectorAll(sel).forEach(function (el, i) {
        var x = el.scrollWidth - el.clientWidth;
        if (x > 1) r.push('XOVER~' + sel + '[' + i + ']~xover=' + (x * mm).toFixed(1) + 'mm~');
      });
    });

  var txt = r.join(' ');
  document.getElementById('probe').textContent = txt;
  console.log(txt);
});
</script>
